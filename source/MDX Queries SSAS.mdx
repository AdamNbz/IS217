-- 1
WITH
SET [Models>=50] AS
  FILTER([DIM MODEL].[Model Name].Hierarchy.Levels(1).Members,
         [Measures].[FACT CAR LISTING Count] >= 50)
SELECT
  { [Measures].[Avg Price], [Measures].[FACT CAR LISTING Count] } ON COLUMNS,
  TOPCOUNT([Models>=50], 10, [Measures].[Avg Price]) ON ROWS
FROM [SSIS];


-- 2
WITH
MEMBER [Measures].[Median Price (by Model)] AS
  Median(
    EXISTING [DIM MODEL].[Model Name].Hierarchy.Levels(1).Members,
    [Measures].[Avg Price]
  )
SELECT
  { [Measures].[Median Price (by Model)] } ON COLUMNS,
  TOPCOUNT([DIM BRAND].[Brand Name].Hierarchy.Levels(1).Members,
           10, [Measures].[Median Price (by Model)]) ON ROWS
FROM [SSIS];


-- 3
WITH
SET [Top10ModelsPerBrand] AS
  GENERATE(
    [DIM BRAND].[Brand Name].Hierarchy.Levels(1).Members,
    { ([DIM BRAND].[Brand Name].CurrentMember)
      * TOPCOUNT(
          EXISTING [DIM MODEL].[Model Name].Hierarchy.Levels(1).Members,
          10, [Measures].[Avg Price])
    }
  )
SELECT { [Measures].[Avg Price] } ON COLUMNS,
       [Top10ModelsPerBrand] ON ROWS
FROM [SSIS];



-- 4
SELECT
  { [Measures].[Avg Price] } ON COLUMNS,
  [DIM AGE BUCKET].[Age Bucket Label].Hierarchy.Levels(1).Members ON ROWS
FROM [SSIS];


-- 5
SELECT
  { [Measures].[Avg Price] } ON COLUMNS,
  NON EMPTY (
    [DIM BRAND].[Brand Name].Hierarchy.Levels(1).Members
  * [DIM FUEL].[Fuel Type].Hierarchy.Levels(1).Members
  ) ON ROWS
FROM [SSIS];


-- 6
SELECT
  { [Measures].[Avg Price per 1k km] } ON COLUMNS,
  TOPCOUNT(
    [DIM MODEL].[Model Name].Hierarchy.Levels(1).Members,
    10, -[Measures].[Avg Price per 1k km]
  ) ON ROWS
FROM [SSIS];


-- 7
SELECT
  { [Measures].[Avg Price] } ON COLUMNS,
  [DIM MILEAGE BUCKET].[Mileage Bucket Label].Hierarchy.Levels(1).Members ON ROWS
FROM [SSIS];


-- 8
WITH
MEMBER [Measures].[Prev Bucket Avg Price] AS
  ([Measures].[Avg Price],
   [DIM AGE BUCKET].[Age Bucket Label].CurrentMember.PrevMember)
MEMBER [Measures].[Depreciation YoY %] AS
  IIF([Measures].[Prev Bucket Avg Price]=0 OR IsEmpty([Measures].[Prev Bucket Avg Price]),
      NULL,
      ([Measures].[Avg Price]-[Measures].[Prev Bucket Avg Price])
      / [Measures].[Prev Bucket Avg Price]),
  FORMAT_STRING="Percent"
SELECT
  { [Measures].[Avg Price], [Measures].[Depreciation YoY %] } ON COLUMNS,
  [DIM AGE BUCKET].[Age Bucket Label].Hierarchy.Levels(1).Members ON ROWS
FROM [SSIS];


-- 9
WITH
SET [TopModelEachBrand] AS
  GENERATE(
    [DIM BRAND].[Brand Name].Hierarchy.Levels(1).Members,
    TOPCOUNT(
      EXISTING ( { [DIM BRAND].[Brand Name].CurrentMember }
                 * [DIM MODEL].[Model Name].Hierarchy.Levels(1).Members ),
      1, [Measures].[FACT CAR LISTING Count])
  )
SELECT
  { [Measures].[FACT CAR LISTING Count] } ON COLUMNS,
  [TopModelEachBrand] ON ROWS
FROM [SSIS];


-- 10
SELECT
  { [Measures].[Market Share %], [Measures].[FACT CAR LISTING Count] } ON COLUMNS,
  ORDER(
    [DIM BRAND].[Brand Name].Hierarchy.Levels(1).Members,
    [Measures].[Market Share %], BDESC
  ) ON ROWS
FROM [SSIS];


-- 11
WITH
SET [Top10ModelsPerFuel] AS
  GENERATE(
    [DIM FUEL].[Fuel Type].Hierarchy.Levels(1).Members,

      * TOPCOUNT(
          EXISTING [DIM MODEL].[Model Name].Hierarchy.Levels(1).Members,

       [Top10ModelsPerFuel] ON ROWS
FROM [SSIS];


-- 12
SELECT
  { [Measures].[Overpriced %], [Measures].[Overpriced Flag] } ON COLUMNS,
  NON EMPTY (
    [DIM BRAND].[Brand Name].Hierarchy.Levels(1).Members
  * [DIM MODEL].[Model Name].Hierarchy.Levels(1).Members
  ) ON ROWS
FROM [SSIS];


-- 13
WITH
SET [ModelsByBrand] AS
  EXISTING [DIM MODEL].[Model Name].Hierarchy.Levels(1).Members
MEMBER [Measures].[P25 (by Model)] AS
  ( [Measures].[Avg Price],
    ( ORDER([ModelsByBrand], [Measures].[Avg Price], BASC)
    ).Item( INT( (COUNT([ModelsByBrand]) - 1) * 0.25 ) ) )
MEMBER [Measures].[P50 (by Model)] AS
  ( [Measures].[Avg Price],
    ( ORDER([ModelsByBrand], [Measures].[Avg Price], BASC)
    ).Item( INT( (COUNT([ModelsByBrand]) - 1) * 0.50 ) ) )
MEMBER [Measures].[P75 (by Model)] AS
  ( [Measures].[Avg Price],
    ( ORDER([ModelsByBrand], [Measures].[Avg Price], BASC)
    ).Item( INT( (COUNT([ModelsByBrand]) - 1) * 0.75 ) ) )
SELECT
  { [Measures].[P25 (by Model)], [Measures].[P50 (by Model)], [Measures].[P75 (by Model)] } ON COLUMNS,
  [DIM BRAND].[Brand Name].Hierarchy.Levels(1).Members ON ROWS
FROM [SSIS];


-- 14
WITH
SET [AllModels] AS [DIM MODEL].[Model Name].Hierarchy.Levels(1).Members
MEMBER [Measures].[P75_AllModels] AS
  ( [Measures].[Avg Price],
    ( ORDER([AllModels], [Measures].[Avg Price], BASC)
    ).Item( INT( (COUNT([AllModels]) - 1) * 0.75 ) ) )
SET [OldModelsRich] AS
  FILTER([DIM MODEL].[Model Name].Hierarchy.Levels(1).Members,
         [Measures].[Avg Price] > [Measures].[P75_AllModels])
SELECT
  { [Measures].[Avg Price] } ON COLUMNS,
  TOPCOUNT([OldModelsRich], 10, [Measures].[Avg Price]) ON ROWS
FROM [SSIS]
WHERE (
  TAIL([DIM AGE BUCKET].[Age Bucket Label].Hierarchy.Levels(1).Members,1).Item(0)
);



-- 15
WITH
SET [BudgetLowMileageModels] AS
  FILTER([DIM MODEL].[Model Name].Hierarchy.Levels(1).Members,
         CoalesceEmpty([Measures].[Avg Price],0) <= 10000
     AND CoalesceEmpty([Measures].[Avg Mileage],9E99) < 50000)
SELECT
  { [Measures].[Avg Price], [Measures].[Avg Mileage], [Measures].[FACT CAR LISTING Count] } ON COLUMNS,
  TOPCOUNT([BudgetLowMileageModels], 10, [Measures].[FACT CAR LISTING Count]) ON ROWS
FROM [SSIS];

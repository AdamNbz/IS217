-- Câu truy vấn 1. Thị phần của mỗi hãng (tính theo số lượng xe).
WITH
SET [BrandsWithData] AS
    FILTER(
        [DIM BRAND].[Brand Name].[Brand Name].ALLMEMBERS,
        NOT ISEMPTY([Measures].[FACT CAR LISTING Count])
        AND NOT ISEMPTY([Measures].[Market Share %])
    )
SELECT 
    { [Measures].[Market Share %], [Measures].[FACT CAR LISTING Count] } ON COLUMNS,
    ORDER( [BrandsWithData], [Measures].[Market Share %], BDESC ) ON ROWS
FROM [SSIS];

--Câu truy vấn 2. So sánh giá trung bình theo nhóm tuổi xe
WITH
SET [AgeBuckets_NoUnknown] AS
    FILTER(
        [DIM AGE BUCKET].[Age Bucket Label].Members,
        UCASE([DIM AGE BUCKET].[Age Bucket Label].CurrentMember.Name) <> "UNKNOWN"
    )
SELECT 
    { [Measures].[Avg Price] } ON COLUMNS,
    ORDER(
        [AgeBuckets_NoUnknown],
        [DIM AGE BUCKET].[Age Bucket Order].MemberValue,
        BASC
    ) ON ROWS
FROM [SSIS];

-- Câu truy vấn 3. So sánh giá trung bình giữa các “bucket” quãng đường (dưới 20k, 20–50k, 50–100k, >100k km)
WITH
SET [MileageBuckets_NoUnknown_NoAll] AS
    FILTER(
        [DIM MILEAGE BUCKET].[Mileage Bucket Label].Members,
        NOT [DIM MILEAGE BUCKET].[Mileage Bucket Label].CurrentMember
            IS [DIM MILEAGE BUCKET].[Mileage Bucket Label].DefaultMember
        AND UCASE(
            [DIM MILEAGE BUCKET].[Mileage Bucket Label].CurrentMember.Name
        ) <> "UNKNOWN"
    )
SELECT 
    { [Measures].[Avg Price] } ON COLUMNS,
    ORDER(
        [MileageBuckets_NoUnknown_NoAll],
        [DIM MILEAGE BUCKET].[Mileage Bucket Order].MemberValue,
        BASC
    ) ON ROWS
FROM [SSIS];

-- Câu truy vấn 4. Khấu hao giá theo tuổi xe (Tuổi xe = Năm hiện tại – Năm sản xuất).
SELECT { [Measures].[Avg Price] } ON COLUMNS,
       ORDER([DIM YEAR].[Manufacture Year].[Manufacture Year].Members,
             [DIM YEAR].[Manufacture Year].MemberValue, BASC) ON ROWS
FROM [SSIS];

-- Câu truy vấn 5. So sánh tổng toàn thị trường với từng hãng.
WITH
SET [BrandPlusAll] AS
    { [DIM BRAND].[Brand Name].DefaultMember }
    + [DIM BRAND].[Brand Name].Members
SELECT
    { [Measures].[FACT CAR LISTING Count],
      [Measures].[Avg Price] } ON COLUMNS,
    [BrandPlusAll] ON ROWS
FROM [SSIS];

-- Câu truy vấn 6. Thông tin chi tiết (giá trung bình và số lượng xe) các model của hãng Toyota.
SELECT
    { [Measures].[Avg Price],
      [Measures].[FACT CAR LISTING Count] } ON COLUMNS,
    NON EMPTY
        [DIM MODEL].[Brand-Model].[Model].Members ON ROWS
FROM [SSIS]
WHERE (
    [DIM BRAND].[Brand Name].[Toyota]
);

-- Câu truy vấn 7. Model xe bán chạy nhất (nhiều xe nhất) trong từng hãng.
WITH
SET [TopModelEachBrand] AS
  GENERATE(
    [DIM BRAND].[Brand Name].[Brand Name].Members,
    [DIM BRAND].[Brand Name].CurrentMember *
    TOPCOUNT(
        EXISTING [DIM MODEL].[Brand-Model].[Model].Members,
        1,
        [Measures].[FACT CAR LISTING Count]
    )
  )
SELECT
    { [Measures].[FACT CAR LISTING Count] } ON COLUMNS,
    NON EMPTY [TopModelEachBrand] ON ROWS
FROM [SSIS];

-- Câu truy vấn 8. Top 10 model xe đắt nhất theo từng loại nhiên liệu xe (sắp xếp theo giá trung bình).
WITH
SET [FuelTypes_NoAll] AS
    FILTER(
        [DIM FUEL].[Fuel Type].Members,
        NOT [DIM FUEL].[Fuel Type].CurrentMember IS [DIM FUEL].[Fuel Type].DefaultMember
    )

SET [Top10ModelsPerFuel] AS
  GENERATE(
    [FuelTypes_NoAll], 
    
    [DIM FUEL].[Fuel Type].CurrentMember * TOPCOUNT(
        EXISTING [DIM MODEL].[Model Name].Members,
      10, 
      [Measures].[Avg Price]
    )
  )

SELECT 
    { [Measures].[Avg Price] } ON COLUMNS,
    NON EMPTY [Top10ModelsPerFuel] ON ROWS
FROM [SSIS];

-- Câu truy vấn 9. Giá trung bình theo (Age bucket × Fuel type) chỉ cho hãng Toyota.
WITH
SET [AgeBuckets_NoUnknown] AS
    FILTER(
        [DIM AGE BUCKET].[Age Bucket Label].Members,
        UCASE(
            [DIM AGE BUCKET].[Age Bucket Label].CurrentMember.Name
        ) <> "UNKNOWN"
        AND NOT [DIM AGE BUCKET].[Age Bucket Label].CurrentMember
            IS [DIM AGE BUCKET].[Age Bucket Label].DefaultMember
    )
SET [FuelTypes_NoAll] AS
    FILTER(
        [DIM FUEL].[Fuel Type].Members,
        NOT [DIM FUEL].[Fuel Type].CurrentMember
            IS [DIM FUEL].[Fuel Type].DefaultMember
    )
SELECT
    NON EMPTY [FuelTypes_NoAll] ON COLUMNS,
    NON EMPTY [AgeBuckets_NoUnknown] ON ROWS
FROM [SSIS]
WHERE (
    [DIM BRAND].[Brand Name].&[Toyota],
    [Measures].[Avg Price]
);

-- Câu truy vấn 10. So sánh giá trung bình Hybrid với Petrol/Diesel trong cùng hãng.
WITH
SET [Brands_NoAll] AS
    FILTER(
        [DIM BRAND].[Brand Name].Members,
        NOT [DIM BRAND].[Brand Name].CurrentMember
            IS [DIM BRAND].[Brand Name].DefaultMember
    )
SELECT
  { [DIM FUEL].[Fuel Type].&[Hybrid],
    [DIM FUEL].[Fuel Type].&[Petrol],
    [DIM FUEL].[Fuel Type].&[Diesel] } ON COLUMNS,
  [Brands_NoAll] ON ROWS
FROM [SSIS]
WHERE ( [Measures].[Avg Price] );

--Câu 11. Top 10 model xe có giá trung bình cao nhất (lọc model ≥ 50 xe).
WITH
SET [Models>=50] AS
  FILTER([DIM MODEL].[Model].Members,
         [Measures].[FACT CAR LISTING Count] >= 50)
SELECT
  { [Measures].[Avg Price], [Measures].[FACT CAR LISTING Count] } ON COLUMNS,
  TOPCOUNT([Models>=50], 10, [Measures].[Avg Price]) ON ROWS
FROM [SSIS];

-- Câu truy vấn 12 Top 10 model xe đắt nhất (sắp xếp theo giá trung bình).
SELECT { [Measures].[Avg Price] } ON COLUMNS,
       TOPCOUNT([DIM MODEL].[Brand-Model].[Model], 10, [Measures].[Avg Price]) ON ROWS
FROM [SSIS];

-- Câu truy vấn 13. Top 10 model xe “giá trị” nhất tính theo giá trung bình trên 1,000 km thấp nhất.
WITH
SET [ModelsWithPricePerKm] AS
    FILTER(
        [DIM MODEL].[Brand-Model].[Model].Members,
        NOT ISEMPTY([Measures].[Avg Price per 1k km])
    )

SELECT 
    { [Measures].[Avg Price per 1k km] } ON COLUMNS,
    TOPCOUNT(
        [ModelsWithPricePerKm],
        10,
        -[Measures].[Avg Price per 1k km]
    ) ON ROWS
FROM [SSIS];

-- Câu truy vấn 14. Giá trung bình, số lượng xe và khấu hao giá thị trường của từng hãng xe.
SELECT
    NON EMPTY
    [DIM BRAND].[Brand Name].Members ON COLUMNS,
    {
        [Measures].[Avg Price],
        [Measures].[FACT CAR LISTING Count],
        [Measures].[Market Share %]
    } ON ROWS
FROM [SSIS];

--Câu 15. Top 10 hãng xe có median của giá trung bình theo model xe cao nhất.
WITH
MEMBER [Measures].[Median Model Avg] AS
    MEDIAN(
        EXISTING [DIM MODEL].[Model].Members,
        [Measures].[Avg Price]
    )

SET [BrandsWithMedian] AS
    FILTER(
        [DIM MODEL].[Brand].Members,
        NOT ISEMPTY([Measures].[Median Model Avg])
    )

SELECT
    { [Measures].[Median Model Avg] } ON COLUMNS,
    TOPCOUNT(
        [BrandsWithMedian],
        10,
        [Measures].[Median Model Avg]
    ) ON ROWS
FROM [SSIS];
